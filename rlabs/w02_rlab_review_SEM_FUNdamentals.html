Setup

library(tidyverse)
library(lavaan)
library(lavaanPlot)

pisa <- read_csv("w01_data_pisa22_USA_cln.csv", show_col_types = FALSE)
pared_mean  <- mean(pisa$PAREDINT, na.rm = TRUE)
books_mean  <- mean(pisa$BOOKSHOME, na.rm = TRUE)

pisa <- pisa |>
  mutate(
    PAREDINT_c  = PAREDINT - pared_mean,
    BOOKSHOME_c = BOOKSHOME - books_mean
  )

Model 1: Simple regression

\[\widehat{\text{Reading}} = b_0 + b_1(\text{ParentEd})\]

m1 <- lm(READ_AVG ~ PAREDINT_c, data = pisa)
